def test_upload_anonimo_bloqueado_e_documentos_isolados_por_empresa(started_app):
    from app.models import PersonAttachment

    a = make_company("Anexos Empresa A")
    b = make_company("Anexos Empresa B")
    ua = make_user("uploads-a@teste.com", company_id=a)
    ub = make_user("uploads-b@teste.com", company_id=b)
    relative = f"persons/person-{ub}-documento.pdf"
    path = config.UPLOAD_DIR / relative
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_bytes(b"%PDF arquivo ficticio e isolado da empresa B")
    db = _db()
    try:
        person = Person(name="Cliente Privado Empresa B", company_id=b, user_id=ub)
        db.add(person)
        db.flush()
        db.add(PersonAttachment(person_id=person.id, file_path="uploads/" + relative,
                                file_name="documento.pdf", file_type="pdf"))
        db.commit()
    finally:
        db.close()

    url = "/uploads/" + relative
    assert new_client(started_app).get(url).status_code == 401
    assert logged(started_app, "uploads-a@teste.com").get(url).status_code == 404
    valid = logged(started_app, "uploads-b@teste.com").get(url)
    assert valid.status_code == 200
    assert valid.content.startswith(b"%PDF")
    assert valid.headers["cache-control"] == "private, no-store"
    assert valid.headers["content-disposition"] == "attachment"
    assert valid.headers["x-content-type-options"] == "nosniff"


def test_uploads_nao_autorizam_arquivo_apenas_por_existir(started_app):
    make_user("uploads-unlisted@teste.com")
    path = config.UPLOAD_DIR / "persons" / "sem-registro.jpg"
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_bytes(b"IMG ficticia")
    client = logged(started_app, "uploads-unlisted@teste.com")
    assert client.get("/uploads/persons/sem-registro.jpg").status_code == 404
    assert client.get("/uploads/unknown/arquivo.jpg").status_code == 404
    assert client.get("/uploads/persons/%2e%2e/segredo.jpg").status_code in {400, 404}


def test_uploads_cotacao_exigem_registro_da_mesma_empresa(started_app):
    a = make_company("Cotacao Anexos A")
    b = make_company("Cotacao Anexos B")
    ua = make_user("quote-files-a@teste.com", company_id=a)
    ub = make_user("quote-files-b@teste.com", company_id=b)
    gid = make_group(ub, b, "Cotacao da Empresa B")
    relative = f"quotes/{gid}/quote-anexo.pdf"
    path = config.UPLOAD_DIR / relative
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_bytes(b"%PDF teste")
    db = _db()
    try:
        db.add(AcceptedQuote(group_id=gid, user_id=ub, company_id=b,
                             extra_json=__import__('json').dumps({'attachments': [
                                 {'path': 'uploads/' + relative, 'id': 'anexo-1'}
                             ]})))
        db.commit()
    finally:
        db.close()
    url = '/uploads/' + relative
    assert logged(started_app, 'quote-files-a@teste.com').get(url).status_code == 404
    assert logged(started_app, 'quote-files-b@teste.com').get(url).status_code == 200


def test_health_db_nao_divulga_exception_details(started_app, monkeypatch):
    import importlib
    app_main = importlib.import_module('app.main')

    class FailedEngine:
        def connect(self):
            raise RuntimeError('postgresql://admin:SENHA_SUPER_SECRETA@host')

    monkeypatch.setattr(app_main, 'engine', FailedEngine())
    response = new_client(started_app).get('/health/db')
    assert response.status_code == 503
    assert response.json() == {'status': 'error', 'database': 'unavailable'}
    assert 'SENHA_SUPER_SECRETA' not in response.text
